# re-implemented on stefi-lib. Bare NUCLEO-G431KB: -DSTEFI_BOARD=nucleo_g431kb
if(NOT STEFI_BOARD STREQUAL "nucleo_g431kb")
    return()
endif()

# g4_example(<name> <dir> [DEFINES ...]) -> executable <name> from <dir>/main.c
function(g4_example name dir)
    cmake_parse_arguments(EX "" "" "DEFINES" ${ARGN})
    add_executable(${name} ${dir}/main.c)
    target_include_directories(${name} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/common)
    target_compile_definitions(${name} PRIVATE ${EX_DEFINES})
    target_link_libraries(${name} PRIVATE stefi::board stefi::simpleio m)
endfunction()
