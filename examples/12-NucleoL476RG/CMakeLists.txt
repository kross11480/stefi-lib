# re-implemented on stefi-lib. Bare NUCLEO-L476RG: -DSTEFI_BOARD=nucleo_l476rg
if(NOT STEFI_BOARD STREQUAL "nucleo_l476rg")
    return()
endif()

# l4_example(<name> <dir> [DEFINES ...]) -> executable <name> from <dir>/main.c.
# A <dir>/firewall.ld, if present, adds the firewall segments to the board script.
function(l4_example name dir)
    cmake_parse_arguments(EX "" "" "DEFINES" ${ARGN})
    add_executable(${name} ${dir}/main.c)
    target_compile_definitions(${name} PRIVATE ${EX_DEFINES})
    target_link_libraries(${name} PRIVATE stefi::board stefi::simpleio m)
    if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/${dir}/firewall.ld)
        target_link_options(${name} PRIVATE
            -Wl,-T,${CMAKE_CURRENT_SOURCE_DIR}/${dir}/firewall.ld)
    endif()
endfunction()
